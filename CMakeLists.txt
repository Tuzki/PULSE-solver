cmake_minimum_required(VERSION 3.24)
project(pulse_solver VERSION 1.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
 set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
file(GLOB_RECURSE PULSE_ID_FILES CONFIGURE_DEPENDS src/*.cpp include/*.hpp include/*.h apps/*.cpp)
list(SORT PULSE_ID_FILES)
set(PULSE_SOURCE_ID_INPUT "")
foreach(PULSE_ID_FILE IN LISTS PULSE_ID_FILES)
 file(SHA256 "${PULSE_ID_FILE}" PULSE_FILE_HASH)
 file(RELATIVE_PATH PULSE_ID_REL "${CMAKE_CURRENT_SOURCE_DIR}" "${PULSE_ID_FILE}")
 string(APPEND PULSE_SOURCE_ID_INPUT "${PULSE_ID_REL}:${PULSE_FILE_HASH}\n")
endforeach()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" PULSE_CMAKE_HASH)
string(APPEND PULSE_SOURCE_ID_INPUT "CMakeLists.txt:${PULSE_CMAKE_HASH}\n")
string(SHA256 PULSE_SOURCE_SHA256 "${PULSE_SOURCE_ID_INPUT}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${PULSE_ID_FILES})
add_library(pulse_core STATIC src/netlist.cpp src/common.cpp src/semantics.cpp src/descriptor.cpp
 src/propagate.cpp src/runtime.cpp src/engineering_flow.cpp src/engineering_mode.cpp
 src/time_origin.cpp src/output.cpp src/npz.cpp src/first_root/first_root.cpp
 src/first_root/interval_math.cpp src/first_root/exclusion.cpp
 src/pre_netlist.cpp src/constitutive.cpp src/software.cpp src/state_coordinates.cpp src/solver_strategy.cpp)
if(WIN32)
 target_link_libraries(pulse_core PUBLIC bcrypt)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
 find_package(OpenSSL REQUIRED COMPONENTS Crypto)
 target_link_libraries(pulse_core PUBLIC ${CMAKE_DL_LIBS} OpenSSL::Crypto)
else()
 message(FATAL_ERROR "Supported native platforms are Windows and Linux")
endif()
target_include_directories(pulse_core PUBLIC include)
target_include_directories(pulse_core SYSTEM PUBLIC third_party third_party/eigen-3.4.0)
target_include_directories(pulse_core PRIVATE include/pulse/first_root)
target_compile_definitions(pulse_core PUBLIC EIGEN_MPL2_ONLY EIGEN_DONT_PARALLELIZE)
set_source_files_properties(src/runtime.cpp PROPERTIES COMPILE_DEFINITIONS PULSE_SOURCE_SHA256="${PULSE_SOURCE_SHA256}")
if(MSVC)
 target_compile_options(pulse_core PUBLIC /fp:strict PRIVATE /W4)
else()
 target_compile_options(pulse_core PUBLIC -ffp-contract=off -fno-fast-math PRIVATE -Wall -Wextra)
endif()
add_executable(pulse-solver apps/cli.cpp)
target_link_libraries(pulse-solver PRIVATE pulse_core)
add_library(pulse_solver SHARED src/abi.cpp)
target_link_libraries(pulse_solver PRIVATE pulse_core)
target_compile_definitions(pulse_solver PRIVATE PULSE_BUILD_SHARED)
set_target_properties(pulse_solver PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
include(CTest)
