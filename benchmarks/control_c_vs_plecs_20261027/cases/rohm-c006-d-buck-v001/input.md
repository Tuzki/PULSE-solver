# PULSE Netlist v1.2pre

## 参数域
```pulse
.PARAM VIN=400
.PARAM VREF=48
.PARAM PLOAD=1000
.PARAM RLOAD={VREF*VREF/PLOAD}
.PARAM LLOAD=470u
.PARAM RL=5m
.PARAM COUT=100u
.PARAM ESR=10m
.PARAM VC0=48
.PARAM IL0=0
.PARAM FS=50k
.PARAM TS={1/FS}
.PARAM DUTY={VREF/VIN}
.PARAM TON={DUTY*TS}
.PARAM MOS_RON=13m
.PARAM D_VF=0.9385
.PARAM D_RON=0.0187725
.PARAM R_OFF=1G
```

## 仿真参数配置
```pulse
.TRAN T_END=300m SAVE_DT=200n
.SAVE VINBUS IIN VOUT IOUT VDS ID IDIODE
```

## 电气域连接网表
```pulse
V_BUS VIN 0 DC={VIN}


.SUBCKT ROHM_MOSFET_WITH_DIODE D S G
SENSOR_VDS VDS V D S
CMP_BODY BODY VDS OP=LT THRESHOLD=0
SEL_ON BOOL::ON G BODY 1
SW_CHANNEL D S ON RON={MOS_RON} ROFF={R_OFF}
.ENDS ROHM_MOSFET_WITH_DIODE

X_Q1 VIN NSW GATE ROHM_MOSFET_WITH_DIODE

.SUBCKT ROHM_SBD A K
R_JUNCTION A K IV_TABLE=[(-1000,{(-1000-D_VF)/R_OFF}),({D_VF},0),(1000,{(1000-D_VF)/D_RON})] RANGE=EXTEND
.ENDS ROHM_SBD
X_D1 0 NSW ROHM_SBD

L_CHOKE NSW N_L L={LLOAD} IC_I={IL0}
R_CHOKE N_L OUT R={RL}
R_ESR OUT NC R={ESR}
C_OUT NC 0 C={COUT} IC_V={VC0}
R_LOAD OUT 0 R={RLOAD}

SENSOR_VIN VINBUS V VIN 0
SENSOR_OUT VOUT V OUT 0
SENSOR_C VC V NC 0
SENSOR_L IL I L_CHOKE
SENSOR_IN I_SOURCE I V_BUS
MATH_IN IIN = -I_SOURCE
SENSOR_LOAD IOUT I R_LOAD
SENSOR_QV VDS V VIN NSW
SENSOR_QI ID I X_Q1/SW_CHANNEL
SENSOR_DI IDIODE I X_D1/R_JUNCTION
```

## 虚拟MCU配置域
```pulse
MCU MCU0 PROGRAM=CTRL
TIMER CONTROL_CLOCK MCU=MCU0 PERIOD={TS} START=0
TASK CONTROL MCU=MCU0 TRIGGER=CONTROL_CLOCK ENTRY=control_step INPUTS=[] OUTPUTS=[BOOL::GATE] INIT_OUTPUTS=[0] PARAMS=[{TON}]
```

## 代码域
```pulse
PROGRAM CTRL SOURCES=["controller.c"]
```
